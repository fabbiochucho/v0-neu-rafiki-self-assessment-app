import { createClient } from "@/lib/supabase/server"

export interface AssessmentDraft {
  id: string
  userId: string
  profileId: string
  domainIds: string[]
  progressData: Record<string, any>
  progressPercentage: number
  lastSavedAt: string
}

export async function saveAssessmentDraft(
  userId: string,
  profileId: string,
  domainIds: string[],
  progressData: Record<string, any>,
  progressPercentage: number
): Promise<AssessmentDraft> {
  const supabase = await createClient()
  
  const { data, error } = await supabase
    .from('assessment_drafts')
    .upsert({
      user_id: userId,
      profile_id: profileId,
      domain_ids: domainIds,
      progress_data: progressData,
      progress_percentage: progressPercentage,
      updated_at: new Date().toISOString()
    })
    .select()
    .single()
    
  if (error) throw error
  return data
}

export async function getAssessmentDraft(
  userId: string,
  profileId: string
): Promise<AssessmentDraft | null> {
  const supabase = await createClient()
  
  const { data } = await supabase
    .from('assessment_drafts')
    .select('*')
    .eq('user_id', userId)
    .eq('profile_id', profileId)
    .single()
    
  return data
}

export async function compareAssessments(
  baselineAssessmentId: string,
  followUpAssessmentId: string,
  userId: string
) {
  const supabase = await createClient()
  
  // Fetch both assessments
  const { data: baseline } = await supabase
    .from('assessments')
    .select('*, assessment_results(*)')
    .eq('id', baselineAssessmentId)
    .single()
    
  const { data: followUp } = await supabase
    .from('assessments')
    .select('*, assessment_results(*)')
    .eq('id', followUpAssessmentId)
    .single()
    
  // Calculate changes
  const improvementAreas: string[] = []
  const declineAreas: string[] = []
  let totalChange = 0
  
  baseline.assessment_results?.forEach((baseResult: any) => {
    const followResult = followUp.assessment_results?.find(
      (r: any) => r.domain_name === baseResult.domain_name
    )
    if (followResult) {
      const change = followResult.percentage_score - baseResult.percentage_score
      totalChange += change
      if (change > 5) improvementAreas.push(baseResult.domain_name)
      if (change < -5) declineAreas.push(baseResult.domain_name)
    }
  })
  
  // Store comparison
  const { data } = await supabase
    .from('assessment_comparisons')
    .insert({
      user_id: userId,
      baseline_assessment_id: baselineAssessmentId,
      follow_up_assessment_id: followUpAssessmentId,
      improvement_areas: improvementAreas,
      decline_areas: declineAreas,
      overall_change_percentage: totalChange / (baseline.assessment_results?.length || 1)
    })
    .select()
    .single()
    
  return data
}

export async function exportAssessmentReport(
  assessmentId: string,
  userId: string,
  format: 'pdf' | 'json' = 'pdf',
  isProfessional: boolean = false
) {
  const supabase = await createClient()
  
  // Generate shareable token
  const token = Math.random().toString(36).substring(2, 15)
  
  const { data } = await supabase
    .from('assessment_exports')
    .insert({
      assessment_id: assessmentId,
      user_id: userId,
      export_format: format,
      is_professional_report: isProfessional,
      shareable_token: token,
      token_expires_at: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString()
    })
    .select()
    .single()
    
  return data
}

export async function getAssessmentResults(assessmentId: string) {
  const supabase = await createClient()
  
  const { data } = await supabase
    .from('assessments')
    .select('*, assessment_results(*), user_profiles(*)')
    .eq('id', assessmentId)
    .single()
    
  return data
}

  const supabase = await createClient()
  
  const { error } = await supabase
    .from('analytics_events')
    .insert(event)
    
  if (error) console.error('Analytics error:', error)
}

export async function getConversionFunnel(
  startDate: string,
  endDate: string,
  country?: string
) {
  const supabase = await createClient()
  
  const events = [
    'landing_page_view',
    'signup_initiated',
    'signup_completed',
    'assessment_started',
    'assessment_completed'
  ]
  
  const funnel = await Promise.all(
    events.map(event =>
      supabase
        .from('analytics_events')
        .select('*', { count: 'exact', head: true })
        .eq('event_type', event)
        .gte('timestamp', startDate)
        .lte('timestamp', endDate)
        .then(({ count }) => ({
          stage: event,
          count: count || 0
        }))
    )
  )
  
  return funnel
}
